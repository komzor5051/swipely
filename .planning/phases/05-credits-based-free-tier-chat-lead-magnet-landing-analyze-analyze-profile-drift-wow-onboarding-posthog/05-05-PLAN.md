      ```typescript
      import { NextRequest, NextResponse } from "next/server";
      import { z } from "zod";
      import { hashIp, checkAnalyzeRateLimit } from "@/lib/services/analyze-rate-limit";

      const BodySchema = z.object({ topic: z.string().min(3).max(300), anonymous_id: z.string().optional() });

      export async function POST(req: NextRequest) {
        let body: { topic: string; anonymous_id?: string };
        try { body = BodySchema.parse(await req.json()); }
        catch { return NextResponse.json({ error: "INVALID_BODY" }, { status: 400 }); }

        // Reuse rate limit, but with prefix so it's separate from /api/analyze
        const ipHash = "preview:" + hashIp(req);
        const rate = await checkAnalyzeRateLimit(ipHash);
        if (!rate.allowed) return NextResponse.json({ error: "RATE_LIMIT", reason: rate.reason }, { status: 429 });

        // Gemini text-only outline. NO PNG render, NO credit debit.
        const proxyUrl = process.env.GEMINI_PROXY_URL;
        const apiKey = process.env.GOOGLE_GEMINI_API_KEY;
        if (!proxyUrl || !apiKey) {
          return NextResponse.json({ error: "GEMINI_NOT_CONFIGURED" }, { status: 500 });
        }
        const prompt = `Сгенерируй структуру 5-слайдовой Instagram карусели на тему: "${body.topic}". Верни ТОЛЬКО JSON-массив из 5 объектов вида {"title":"...","bullets":["...","...","..."]}. Без markdown.`;
        // Use existing Gemini proxy pattern from lib/services/profile-recommendations.ts (callGeminiJsonWithRetry).
        // If that helper isn't easily importable, inline a minimal fetch + JSON.parse with markdown stripping.
        try {
          const res = await fetch(`${proxyUrl}/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ contents: [{ role: "user", parts: [{ text: prompt }] }], generationConfig: { responseMimeType: "application/json" } }),
          });
          const data = await res.json();
          const text = data?.candidates?.[0]?.content?.parts?.[0]?.text ?? "[]";
          const cleaned = text.replace(/^```json\s*/m, "").replace(/```\s*$/m, "").trim();
          const slides = JSON.parse(cleaned);
          if (!Array.isArray(slides) || slides.length === 0) {
            return NextResponse.json({ error: "GEMINI_BAD_OUTPUT" }, { status: 502 });
          }
          return NextResponse.json({ slides: slides.slice(0, 5) });
        } catch (e) {
          const msg = e instanceof Error ? e.message : "UNKNOWN";
          return NextResponse.json({ error: "PREVIEW_FAILED", detail: msg }, { status: 502 });
        }
      }
      ```
