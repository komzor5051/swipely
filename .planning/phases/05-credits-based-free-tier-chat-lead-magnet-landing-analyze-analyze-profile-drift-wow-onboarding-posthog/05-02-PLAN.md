      ```sql
      -- Credits unification — Phase 5
      alter table profiles add column if not exists credits_balance integer not null default 0;
      alter table profiles add column if not exists credits_granted_at timestamptz;

      -- Backfill: existing users get (tier monthly grant - standard_used) + photo_slides_balance, floored at 0
      update profiles set credits_balance = greatest(0,
        case subscription_tier
          when 'free' then 3
          when 'start' then 30
          when 'blogger' then 30
          when 'creator' then 100
          when 'pro' then 100
          else 3
        end - coalesce(standard_used, 0)
      ) + coalesce(photo_slides_balance, 0)
      where credits_balance = 0;

      -- Extend claim_generation_slot to accept credit cost and decrement credits_balance atomically.
      -- Preserves existing cooldown/daily-limit/standard_used logic (dual-write for 30d).
      create or replace function claim_generation_slot(
        p_user_id uuid,
        p_credit_cost integer default 1
      )
      returns table(allowed boolean, balance_after integer, reason text, wait_seconds integer)
      language plpgsql
      security definer
      as $$
      declare
        v_balance integer;
      begin
        select credits_balance into v_balance from profiles where id = p_user_id for update;
        if v_balance is null then
          return query select false, 0, 'NO_PROFILE'::text, 0;
          return;
        end if;
        if v_balance < p_credit_cost then
          return query select false, v_balance, 'INSUFFICIENT_CREDITS'::text, 0;
          return;
        end if;
        update profiles
          set credits_balance = credits_balance - p_credit_cost,
              standard_used = coalesce(standard_used, 0) + 1
          where id = p_user_id;
        return query select true, v_balance - p_credit_cost, null::text, 0;
      end;
      $$;

      -- Helper RPC for adding credits (webhook add_photo_slides migration target)
      create or replace function add_credits(p_user_id uuid, p_amount integer)
      returns integer
      language plpgsql
      security definer
      as $$
      declare v_new integer;
      begin
        update profiles set credits_balance = credits_balance + p_amount
          where id = p_user_id returning credits_balance into v_new;
        return v_new;
      end;
      $$;

      -- Monthly grant RPC — idempotent within calendar month
      create or replace function grant_monthly_credits(p_user_id uuid)
      returns table(granted integer, new_balance integer)
      language plpgsql
      security definer
      as $$
      declare
        v_tier text;
        v_last timestamptz;
        v_grant integer;
      begin
        select subscription_tier, credits_granted_at into v_tier, v_last from profiles where id = p_user_id;
        if v_last is not null and date_trunc('month', v_last) = date_trunc('month', now()) then
          return query select 0, (select credits_balance from profiles where id = p_user_id);
          return;
        end if;
        v_grant := case v_tier
          when 'free' then 3 when 'start' then 30 when 'blogger' then 30
          when 'creator' then 100 when 'pro' then 100 else 3 end;
        update profiles
          set credits_balance = credits_balance + v_grant,
              credits_granted_at = now(),
              standard_used = 0
          where id = p_user_id;
        return query select v_grant, (select credits_balance from profiles where id = p_user_id);
      end;
      $$;
      ```

    Step 2: Create `swipely-nextjs/lib/billing/credits.ts` exporting:
      ```typescript
      import { createAdminClient } from "@/lib/supabase/admin";

      export type DebitResult = { allowed: boolean; balance_after: number; reason: string | null };

      export async function debitCredits(userId: string, amount: number = 1): Promise<DebitResult> {
        const admin = createAdminClient();
        const { data, error } = await admin.rpc("claim_generation_slot", { p_user_id: userId, p_credit_cost: amount });
        if (error) throw error;
        const row = Array.isArray(data) ? data[0] : data;
        return { allowed: !!row?.allowed, balance_after: row?.balance_after ?? 0, reason: row?.reason ?? null };
      }

      export async function addCredits(userId: string, amount: number): Promise<number> {
        const admin = createAdminClient();
        const { data, error } = await admin.rpc("add_credits", { p_user_id: userId, p_amount: amount });
        if (error) throw error;
        return data as number;
      }

      export async function grantMonthlyCredits(userId: string): Promise<{ granted: number; new_balance: number }> {
        const admin = createAdminClient();
        const { data, error } = await admin.rpc("grant_monthly_credits", { p_user_id: userId });
        if (error) throw error;
        const row = Array.isArray(data) ? data[0] : data;
        return { granted: row?.granted ?? 0, new_balance: row?.new_balance ?? 0 };
      }
      ```
