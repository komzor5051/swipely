      ```sql
      -- Analyze lead-magnet rate limit table
      create table if not exists analyze_rate_limits (
        ip_hash text not null,
        day date not null,
        count integer not null default 0,
        primary key (ip_hash, day)
      );

      create index if not exists analyze_rate_limits_day_idx on analyze_rate_limits(day);

      create or replace function check_analyze_rate_limit(p_ip_hash text)
      returns table(allowed boolean, remaining_today integer, reason text)
      language plpgsql
      security definer
      as $$
      declare
        v_today_count integer;
        v_week_count integer;
        v_daily_cap constant integer := 3;
        v_weekly_cap constant integer := 10;
      begin
        select coalesce(count, 0) into v_today_count
          from analyze_rate_limits where ip_hash = p_ip_hash and day = current_date;
        if v_today_count is null then v_today_count := 0; end if;

        select coalesce(sum(count), 0) into v_week_count
          from analyze_rate_limits
          where ip_hash = p_ip_hash and day >= (current_date - interval '6 days');

        if v_today_count >= v_daily_cap then
          return query select false, 0, 'DAILY_LIMIT'::text;
          return;
        end if;
        if v_week_count >= v_weekly_cap then
          return query select false, 0, 'WEEKLY_LIMIT'::text;
          return;
        end if;

        insert into analyze_rate_limits(ip_hash, day, count) values (p_ip_hash, current_date, 1)
          on conflict (ip_hash, day) do update set count = analyze_rate_limits.count + 1;

        return query select true, (v_daily_cap - v_today_count - 1), null::text;
      end;
      $$;
      ```

    Step 2: Create `swipely-nextjs/lib/services/analyze-rate-limit.ts`:
      ```typescript
      import { createHash } from "crypto";
      import { createAdminClient } from "@/lib/supabase/admin";
      import { NextRequest } from "next/server";

      export function hashIp(req: NextRequest): string {
        const fwd = req.headers.get("x-forwarded-for") ?? "";
        const ip = fwd.split(",")[0]?.trim() || req.headers.get("x-real-ip") || "unknown";
        const salt = process.env.IP_HASH_SALT ?? "swipely-analyze";
        return createHash("sha256").update(ip + ":" + salt).digest("hex").slice(0, 32);
      }

      export type RateLimitResult = { allowed: boolean; remaining_today: number; reason: string | null };

      export async function checkAnalyzeRateLimit(ipHash: string): Promise<RateLimitResult> {
        const admin = createAdminClient();
        const { data, error } = await admin.rpc("check_analyze_rate_limit", { p_ip_hash: ipHash });
        if (error) throw error;
        const row = Array.isArray(data) ? data[0] : data;
        return { allowed: !!row?.allowed, remaining_today: row?.remaining_today ?? 0, reason: row?.reason ?? null };
      }
      ```

      ```typescript
      import { NextRequest, NextResponse } from "next/server";
      import { z } from "zod";
      import { scrapeProfile, detectPlatform } from "@/lib/services/profile-scraper";
      import { generateProfileRecommendations } from "@/lib/services/profile-recommendations";
      import { hashIp, checkAnalyzeRateLimit } from "@/lib/services/analyze-rate-limit";
      import { captureAnalyzeCompleted } from "@/lib/analytics/server";

      const BodySchema = z.object({ url: z.string().url(), anonymous_id: z.string().optional() });

      export async function POST(req: NextRequest) {
        let body: { url: string; anonymous_id?: string };
        try {
          body = BodySchema.parse(await req.json());
        } catch {
          return NextResponse.json({ error: "INVALID_URL" }, { status: 400 });
        }

        const ipHash = hashIp(req);
        const rate = await checkAnalyzeRateLimit(ipHash);
        if (!rate.allowed) {
          return NextResponse.json({ error: "RATE_LIMIT", reason: rate.reason }, { status: 429 });
        }

        const platform = detectPlatform(body.url);
        if (platform === "unknown") {
          return NextResponse.json({ error: "UNSUPPORTED_PLATFORM" }, { status: 400 });
        }

        try {
          const profile = await scrapeProfile(body.url);
          const recs = await generateProfileRecommendations(profile);
          const distinctId = body.anonymous_id || `ip_${ipHash}`;
          captureAnalyzeCompleted(distinctId, {
            platform,
            followers: profile.followers ?? null,
            source: "lead_magnet",
          });
          return NextResponse.json({
            ...recs,
            platform,
            handle: profile.handle ?? null,
            followers: profile.followers ?? null,
          });
        } catch (err) {
          const msg = err instanceof Error ? err.message : "SCRAPE_FAILED";
          return NextResponse.json({ error: "SCRAPE_FAILED", detail: msg }, { status: 502 });
        }
      }
      ```

      ```typescript
      "use client";
      import { useState } from "react";
      import posthog from "posthog-js";
      import { ResultView, type AuditResult } from "./ResultView";

      function getAnonymousId(): string {
        if (typeof window === "undefined") return "";
        let id = localStorage.getItem("swipely_anon_id");
        if (!id) {
          id = crypto.randomUUID();
          localStorage.setItem("swipely_anon_id", id);
        }
        return id;
      }

      export default function AnalyzePage() {
        const [url, setUrl] = useState("");
        const [loading, setLoading] = useState(false);
        const [error, setError] = useState<string | null>(null);
        const [result, setResult] = useState<AuditResult | null>(null);

        async function onSubmit(e: React.FormEvent) {
          e.preventDefault();
          setLoading(true); setError(null);
          const anonId = getAnonymousId();
          try {
            const host = (() => { try { return new URL(url).hostname; } catch { return ""; } })();
            posthog.capture("analyze_started", { url_host: host });
            const res = await fetch("/api/analyze", {
              method: "POST",
              headers: { "Content-Type": "application/json" },
              body: JSON.stringify({ url, anonymous_id: anonId }),
            });
            const data = await res.json();
            if (!res.ok) {
              setError(data.error === "RATE_LIMIT" ? "Превышен дневной лимит анализов. Попробуйте завтра." : "Не удалось проанализировать профиль. Проверьте ссылку.");
              return;
            }
            setResult(data);
            localStorage.setItem("swipely_last_analyze", JSON.stringify(data));
          } catch {
            setError("Сетевая ошибка. Попробуйте ещё раз.");
          } finally {
            setLoading(false);
          }
        }

        return (
          <main className="mx-auto max-w-3xl px-6 py-16">
            <h1 className="text-3xl font-semibold tracking-tight text-neutral-900 dark:text-neutral-50">
              Бесплатный аудит профиля
            </h1>
            <p className="mt-3 text-neutral-600 dark:text-neutral-400">
              Вставьте ссылку на Instagram, Threads, LinkedIn или Telegram-канал — получите анализ и идеи для контента.
            </p>
            <form onSubmit={onSubmit} className="mt-8 flex gap-3">
              <input
                type="url" required value={url} onChange={(e) => setUrl(e.target.value)}
                placeholder="https://instagram.com/username"
                className="flex-1 rounded border border-neutral-300 bg-white px-4 py-3 text-neutral-900 placeholder-neutral-400 focus:border-neutral-900 focus:outline-none dark:border-neutral-700 dark:bg-neutral-900 dark:text-neutral-50"
              />
              <button
                type="submit" disabled={loading}
                className="rounded bg-neutral-900 px-6 py-3 font-medium text-white hover:bg-neutral-800 disabled:opacity-50 dark:bg-neutral-50 dark:text-neutral-900"
              >
                {loading ? "Анализирую…" : "Проанализировать"}
              </button>
            </form>
            {error && <p className="mt-4 text-sm text-red-600">{error}</p>}
            {result && <ResultView result={result} />}
          </main>
        );
      }
      ```
