"use client";

import { useState, useEffect } from "react";
import { X } from "lucide-react";
import { ProButton } from "@/app/pricing/ProButton";

// Last second of February 2026
const PROMO_END = new Date(2026, 1, 28, 23, 59, 59);

function useCountdown(target: Date) {
  const [ms, setMs] = useState(() => Math.max(0, target.getTime() - Date.now()));

  useEffect(() => {
    const id = setInterval(() => {
      setMs(Math.max(0, target.getTime() - Date.now()));
    }, 1000);
    return () => clearInterval(id);
  }, [target]);

  const totalSecs = Math.floor(ms / 1000);
  return {
    days: Math.floor(totalSecs / 86400),
    hours: Math.floor((totalSecs % 86400) / 3600),
    minutes: Math.floor((totalSecs % 3600) / 60),
    seconds: totalSecs % 60,
    expired: ms === 0,
  };
}

function TimerUnit({ value, label }: { value: number; label: string }) {
  return (
    <div className="flex flex-col items-center gap-1">
      <div className="bg-white/10 rounded-xl px-3 py-2 min-w-[52px] text-center">
        <span
          className="text-3xl font-black text-white tabular-nums"
          style={{ fontFamily: "var(--font-mono)" }}
        >
          {String(value).padStart(2, "0")}
        </span>
      </div>
      <span className="text-[10px] text-white/35 uppercase tracking-wider">{label}</span>
    </div>
  );
}

export function PromoPopup() {
  const [open, setOpen] = useState(false);
  const countdown = useCountdown(PROMO_END);

  useEffect(() => {
    if (!sessionStorage.getItem("promo_50_dismissed")) {
      // Small delay so page renders first
      const t = setTimeout(() => setOpen(true), 400);
      return () => clearTimeout(t);
    }
  }, []);

  const handleClose = () => {
    sessionStorage.setItem("promo_50_dismissed", "1");
    setOpen(false);
  };

  if (!open) return null;

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
      {/* Backdrop */}
      <div
        className="absolute inset-0 bg-black/70 backdrop-blur-sm"
        onClick={handleClose}
      />

      {/* Modal */}
      <div className="relative z-10 w-full max-w-md rounded-3xl bg-[#0D0D14] border border-[#D4F542]/20 p-8 shadow-2xl overflow-hidden">
        {/* Glow */}
        <div className="absolute -top-24 -right-24 w-64 h-64 rounded-full bg-[#D4F542]/12 blur-3xl pointer-events-none" />
        <div className="absolute -bottom-20 -left-20 w-56 h-56 rounded-full bg-[#0A84FF]/10 blur-3xl pointer-events-none" />

        {/* Close */}
        <button
          onClick={handleClose}
          className="absolute top-4 right-4 w-8 h-8 rounded-full bg-white/8 hover:bg-white/15 flex items-center justify-center transition-colors z-10"
          aria-label="Закрыть"
        >
          <X className="h-4 w-4 text-white/60" />
        </button>

        {/* Live badge */}
        <div className="relative z-10 inline-flex items-center gap-2 bg-[#D4F542]/12 border border-[#D4F542]/25 px-3 py-1 rounded-full mb-5">
          <span className="w-1.5 h-1.5 rounded-full bg-[#D4F542] animate-pulse" />
          <span className="text-xs font-bold text-[#D4F542] uppercase tracking-wider">
            Только до конца месяца
          </span>
        </div>

        {/* Headline */}
        <div className="relative z-10 mb-6">
          <h2
            className="text-6xl font-black text-[#D4F542] leading-none mb-1"
            style={{ fontFamily: "var(--font-mono)" }}
          >
            −50%
          </h2>
          <p className="text-white/50 text-sm">на все PRO подписки</p>
        </div>

        {/* Price grid */}
        <div className="relative z-10 grid grid-cols-2 gap-3 mb-6">

          <div className="bg-white/5 rounded-2xl p-4 border border-white/8">
            <p className="text-[10px] text-white/30 uppercase tracking-wider mb-3">Месяц</p>
            <span
              className="text-sm text-white/25 line-through block mb-0.5"
              style={{ fontFamily: "var(--font-mono)" }}
            >
              990₽
            </span>
            <div className="flex items-baseline gap-1">
              <span
                className="text-2xl font-black text-white"
                style={{ fontFamily: "var(--font-mono)" }}
              >
                495
              </span>
              <span className="text-white/45 text-sm">₽/мес</span>
            </div>
          </div>

          {/* Yearly */}
          <div className="bg-white/5 rounded-2xl p-4 border border-[#D4F542]/15">
            <div className="flex items-center justify-between mb-3">
              <p className="text-[10px] text-white/30 uppercase tracking-wider">Год</p>
              <span className="text-[9px] font-bold bg-[#D4F542]/15 text-[#D4F542] px-1.5 py-0.5 rounded-full leading-none">
                + −17%

            </div>

            </span>

              <span
                className="text-2xl font-black text-[#D4F542]"
                style={{ fontFamily: "var(--font-mono)" }}

              <span className="text-[#D4F542]/50 text-sm">₽/год</span>

          {countdown.expired ? (
            <p className="text-center text-sm text-white/40">Акция завершена</p>
          ) : (
            <div className="flex items-center justify-center gap-2">
              <TimerUnit value={countdown.days} label="дней" />
              <span className="text-white/25 text-2xl font-black mb-5">:</span>
              <TimerUnit value={countdown.hours} label="часов" />
              <span className="text-white/25 text-2xl font-black mb-5">:</span>
              <TimerUnit value={countdown.minutes} label="минут" />

          <ProButton
            productId="pro_yearly"

            className="w-full rounded-2xl bg-[#D4F542] hover:bg-[#c8e83a] text-[#0D0D14] font-black border-0"
          />
          <ProButton
            productId="pro_monthly"
            label="Месяц — 495₽  →"
            className="w-full rounded-2xl bg-white/8 hover:bg-white/12 text-white border border-white/10 font-semibold"
          />
        </div>
      </div>
    </div>
  );
}

// ── Compact inline countdown for use inside plan cards ──────────────────
export function PromoCountdown() {
  const countdown = useCountdown(PROMO_END);
  if (countdown.expired) return null;
  return (
    <span className="text-xs text-white/35 tabular-nums" style={{ fontFamily: "var(--font-mono)" }}>
      {String(countdown.days).padStart(2, "0")}д{" "}
      {String(countdown.hours).padStart(2, "0")}:
      {String(countdown.minutes).padStart(2, "0")}:
      {String(countdown.seconds).padStart(2, "0")}
    </span>
  );
}
